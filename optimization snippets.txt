// 7. Defer JavaScript
// Excludes jQuery, MapPress, and Google Maps from deferring.
// On pages with a MapPress map, ALL deferring is disabled
// because Google Maps is injected dynamically at runtime
// and cannot be caught by handle-based exclusions.
// ============================================================
function defer_parsing_of_js($tag, $handle) {

    if (is_user_logged_in()) return $tag;

    // Disable all deferring on pages containing a MapPress map
    global $post;
    if (isset($post) && (
        has_shortcode($post->post_content, 'mappress') ||
        has_block('mappress/map', $post) ||
        has_block('mappress/mappress', $post)
    )) {
        return $tag;
    }

    // Handles that must never be deferred
    $no_defer = array(
        'jquery', 'jquery-core', 'jquery-migrate', 'jquery-ui-core',
        'wp-polyfill', 'regenerator-runtime', 'wp-i18n',
        'wp-embed', 'underscore', 'backbone',
        'mappress', 'mappress-pro', 'mappress-maps',
        'google-maps', 'googlemaps', 'maps',
    );

    if (in_array($handle, $no_defer, true)) return $tag;

    // Never defer any Google Maps API script by URL
    if (strpos($tag, 'maps.googleapis.com') !== false ||
        strpos($tag, 'maps.google.com') !== false) {
        return $tag;
    }

    // Defer everything else
    if (strpos($tag, '.js') !== false &&
        strpos($tag, 'defer') === false &&
        strpos($tag, 'async') === false) {
        return str_replace(' src', ' defer="defer" src', $tag);
    }

    return $tag;
}
add_filter('script_loader_tag', 'defer_parsing_of_js', 10, 2);

// 8. Disable Self Pingbacks
// ============================================================
function disable_self_pingbacks(&$links) {
    foreach ($links as $l => $link) {
        if (strpos($link, home_url()) === 0) {
            unset($links[$l]);
        }
    }
}
add_action('pre_ping', 'disable_self_pingbacks');

// 10. Enforce HTTPS

add_action('template_redirect', function() {
    if (!is_ssl() && !is_admin() && !headers_sent()) {
        wp_redirect('https://' . $_SERVER['HTTP_HOST'] . $_SERVER['REQUEST_URI'], 301);
        exit;
    }
});

// ============================================================
// 11. REST API Restriction — MapPress compatible
//
// Reads route from REQUEST_URI (always available) instead of
// query_vars which is empty when this filter fires.
// MapPress REST endpoints are explicitly allowed for guests
// so maps load correctly for all non-logged-in visitors.
// ============================================================
add_filter('rest_authentication_errors', function($result) {

    if (is_user_logged_in()) return $result;

    // Read route directly from REQUEST_URI
    $request_uri = isset($_SERVER['REQUEST_URI']) ? $_SERVER['REQUEST_URI'] : '';
    $rest_prefix = rest_get_url_prefix();
    $route       = '';

    if (!empty($request_uri) && strpos($request_uri, '/' . $rest_prefix . '/') !== false) {
        $route = preg_replace('#^.*/' . preg_quote($rest_prefix, '#') . '/#', '/', $request_uri);
        $route = strtok($route, '?');
    }

    // Not a REST request — don't interfere
    if (empty($route)) return $result;

    // Routes allowed for guest visitors
    $allowed_routes = array(
        '/mappress/',           // MapPress map & POI data
        '/wp/v2/posts',         // MapPress mashup support
        '/wp/v2/pages',         // MapPress mashup support
        '/forminator/',
        '/contact-form-7/',
        '/wpforms/',
        '/gravityforms/',
        '/elementor/',
        '/wc/',
        '/wp/v2/media',
    );

    foreach ($allowed_routes as $allowed) {
        if (strpos($route, $allowed) !== false) {
            return $result;
        }
    }

    return new WP_Error(
        'rest_forbidden',
        __('REST API access is restricted. Please log in.', 'textdomain'),
        array('status' => 401)
    );

});

// 12. Sanitize & Validate Custom AJAX Requests
// ============================================================
function sanitize_api_data($data) {
    if (is_array($data)) {
        return array_map('sanitize_text_field', $data);
    }
    return sanitize_text_field($data);
}

add_action('wp_ajax_nopriv_my_custom_api_action', 'handle_my_custom_api_action');
add_action('wp_ajax_my_custom_api_action',        'handle_my_custom_api_action');

function handle_my_custom_api_action() {
    if (!isset($_REQUEST['nonce']) ||
        !wp_verify_nonce($_REQUEST['nonce'], 'my_custom_action')) {
        wp_send_json_error(array('message' => 'Invalid security token.'), 403);
    }
    $input_data = isset($_GET['data']) ? sanitize_api_data($_GET['data']) : array();
    wp_send_json_success(array('data' => $input_data));
}

// 15. Disable User Enumeration
// Uses 'wp' hook so MapPress rewrite rules register first
// ============================================================
add_action('wp', 'disable_user_enumeration');
function disable_user_enumeration() {
    if (is_admin() || !isset($_SERVER['REQUEST_URI'])) return;

    $is_enum = preg_match('/author=\d+/i', $_SERVER['REQUEST_URI']) ||
               (isset($_REQUEST['author']) && is_numeric($_REQUEST['author']));

    if ($is_enum && !headers_sent()) {
        wp_redirect(home_url(), 301);
        exit;
    }
}
